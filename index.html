const SHEET_NAME = "Saraban"; const FOLDER_ID = "ใส่_ID_โฟลเดอร์_Google_Drive_ของคุณที่นี่"; // ก๊อปปี้ ID จาก URL ของโฟลเดอร์ใน Drive มาใส่ function doGet(e) { const action = e.parameter.action; if (action === "read") { return json(readData()); } return json({ message: "Ready" }); } function doPost(e) { const body = JSON.parse(e.postData.contents); const action = body.action; if (action === "create" || action === "update") { return json(saveData(body.data, action)); } else if (action === "updateStatus") { updateStatus(body.id, body.status); return json({ success: true }); } else if (action === "delete") { deleteData(body.id); return json({ success: true }); } else if (action === "upload") { return json(uploadFile(body)); } } function uploadFile(obj) { const folder = DriveApp.getFolderById(FOLDER_ID); const blob = Utilities.newBlob(Utilities.base64Decode(obj.base64), obj.type, obj.name); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return { url: file.getUrl() }; } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(["id", "receiveNo", "bookDate", "receivedDate", "fromOrg", "subject", "owner", "ownerReceivedDate", "status", "fileUrl"]); } return sh; } function readData() { const sh = getSheet(); const values = sh.getDataRange().getValues(); if (values.length <= 1) return []; const headers = values[0]; return values.slice(1).map(row => { let obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }).reverse(); } function saveData(data, action) { const sh = getSheet(); const values = sh.getDataRange().getValues(); const rowData = [data.id, data.receiveNo, data.bookDate, data.receivedDate, data.fromOrg, data.subject, data.owner, data.ownerReceivedDate, data.status, data.fileUrl || ""]; if (action === "create") { sh.appendRow(rowData); } else { for (let i = 1; i < values.length; i++) { if (values[i][0] === data.id) { sh.getRange(i + 1, 1, 1, rowData.length).setValues([rowData]); break; } } } return { success: true }; } function updateStatus(id, status) { const sh = getSheet(); const values = sh.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (values[i][0] === id) { sh.getRange(i + 1, 9).setValue(status); break; } } } function deleteData(id) { const sh = getSheet(); const values = sh.getDataRange().getValues(); for (let i = values.length - 1; i >= 1; i--) { if (values[i][0] === id) { sh.deleteRow(i + 1); break; } } } function json(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); }
